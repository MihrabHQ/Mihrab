Debian, Ubuntu, Mint (apt)

انسٹال کریں

sudo curl -fsSLo /usr/share/keyrings/mihrab.gpg https://mihrabhq.github.io/packages/mihrab.gpg
sudo curl -fsSLo /etc/apt/sources.list.d/mihrab.sources https://mihrabhq.github.io/packages/mihrab.sources
sudo apt update && sudo apt install mihrab

ترتیبات سمیت ہٹائیں

sudo apt purge mihrab && rm -rf ~/.config/Mihrab
Fedora (dnf)

sudo dnf install mihrab

openSUSE (zypper)

انسٹال کریں

ترتیبات سمیت ہٹائیں

Arch, Manjaro, EndeavourOS (pacman, yay)

انسٹال کریں

curl -fsSL https://mihrabhq.github.io/packages/mihrab.asc | sudo pacman-key --add -
sudo pacman-key --lsign-key 28BB36D6D8388A8EF15EEA24E19DCF135A3D8035
printf '\n[mihrab]\nServer = https://mihrabhq.github.io/packages/arch/$arch\n' | sudo tee -a /etc/pacman.conf
sudo pacman -Syu mihrab

ترتیبات سمیت ہٹائیں

sudo pacman -R mihrab && rm -rf ~/.config/Mihrab

نماز

اگلی نماز کی الٹی گنتی کے ساتھ نماز کے اوقات، ہر نماز کی اذان، قبلہ کمپاس، ہوم اسکرین وجٹس — اور ہر نماز ایک ٹیپ میں درج۔ الارم گھڑی کی طرح پوری اسکرین کا نماز الرٹ، ہر نماز کے لیے اذان، الرٹ یا خاموشی کا انتخاب، اور اینڈرائیڈ پر خود نماز کے دوران خاموش فون۔ اسنوز صرف اتنا ملتا ہے جتنا نماز کا وقت باقی ہو، وقت ختم ہونے سے پہلے ایک «آخری موقع» تک۔ اوقات دیانت، اسلامسکا فوربوندیت، مراکش اور الجزائر کی وزارتوں اور Kemenag کے شائع شدہ اوقات سے چھپے ہوئے منٹ تک مطابق، فون پر حساب؛ MUIS، JAKIM اور Türkiye Takvimi جیسے شائع ہوئے؛ باقی جگہ AlAdhan۔ مختصر موڈ میں آج کے صفحے پر صرف آسمان اور الٹی گنتی رہ جاتی ہے؛ دن کے اوقات کے لیے اوپر سوائپ کریں، اور پورے صفحے کی طرح دن بدن سوائپ کریں۔

پڑھیں اور سنیں

چار روایتوں میں مکمل مدنی مصحف، حفص اور ورش کے لیے رنگین تجوید، اور لفظ خواں: کسی لفظ کو دبائے رکھیں، وہی ایک لفظ پڑھ کر سنایا جائے گا۔ 59 قاری (ورش، قالون اور شعبہ کی روایات سمیت)، اسکرین بند ہونے پر سورت کے بعد سورت تلاوت، تفسیر، تراجم اور ختم کا منصوبہ۔ تلاوت آپ کی جگہ یاد رکھتی ہے اور بتاتی ہے کہ اس وقت سنت کیا پڑھنے کی ترغیب دیتی ہے۔ اور پندرہ سطری IndoPak مصحف، حفص پڑھنے کا دوسرا طریقہ۔

حفظ

جو کچھ آپ کو زبانی یاد ہے اسے نشان زد کریں: تمام 114 سورتوں کے گرڈ پر پوری سورتیں — تیسواں اور اُنتیسواں پارہ ایک ایک ٹیپ میں — یا قاری ہی سے کوئی حصہ: ”یہاں تک حفظ ہے“۔ ہر حصے کا اپنا دور کا شیڈول ہوتا ہے: یاد کرنے کے ایک دن بعد، پھر تین، سات، چودہ اور تیس دن بعد — اور کبھی اُس حد سے زیادہ نہیں جو آپ خود مقرر کریں: ایک ہفتہ، دو ہفتے یا ایک مہینہ۔ ہر سنانے کے بعد آپ بتاتے ہیں کہ کیسا رہا: ”مدد درکار تھی“ حصے کو دوبارہ روزانہ پر لے آتا ہے، ”ٹھیک“ ایک قدم آگے بڑھاتا ہے، اور ”روانی سے“ ایک قدم چھوڑ دیتا ہے۔ آج کا دور اُتنی ہی لمبی فہرست ہے جتنی آپ نے طے کی، روزانہ یاد دہانی کے ساتھ، اور تیس خانے دکھاتے ہیں کہ ہر پارے کا کتنا حصہ یاد ہے۔

زیادہ تر ایپس ایک عمومی طریقہ استعمال کرتی ہیں اور اس جدول سے ایک منٹ یا زیادہ ہٹ جاتی ہیں جس پر آپ کی مسجد چلتی ہے۔ جہاں دیانت، مراکش اور الجزائر کی وزارتوں، Kemenag یا سویڈن کا جدول رائج ہے، وہاں محراب چھپا ہوا منٹ دیتا ہے: ادارے کا اپنا طریقہ، جو دن بہ دن اس کے شائع شدہ جدول سے ملایا گیا ہے۔ باقی جگہوں پر یہ AlAdhan یا آپ کا منتخب کردہ طریقہ استعمال کرتا ہے۔

جدول کی آخری شائع شدہ تاریخ کے بعد کیا ہوتا ہے؟

طریقہ اپنی تصحیحی فہرست کے بغیر اکیلا چلتا رہتا ہے۔ جن اوقات سے اسے ملایا گیا، ان میں طریقہ اکیلا بھی ایک چھوٹے سے حصے کے سوا چھپے ہوئے منٹ پر ہوتا ہے، اور وہ حصہ ایک منٹ ہٹتا ہے، اس سے زیادہ کبھی نہیں؛ ایپ «آج» کے صفحے پر یہ بتا دیتی ہے۔ جب اگلے مہینے شائع ہو کر جانچ لیے جاتے ہیں تو نئی فہرست بغیر اپ ڈیٹ کے ایپ تک پہنچ جاتی ہے۔

میں اوقات خود کیسے جانچ سکتا ہوں؟

محراب کے ماہانہ منظر کا ایک ہفتہ اپنے ادارے کے چھپے ہوئے جدول سے ملائیں۔ طریقے، ہر وقت کو جانچنے والا اسکرپٹ اور اس کا لکھا ہوا ریکارڈ GitHub پر mihrab-methods کے ذخیرے میں سب کے لیے کھلے ہیں۔

کیا محراب دیانت یا کسی وزارت سے وابستہ ہے؟

نہیں۔ محراب آزاد ہے اور جس ادارے کے جدول سے مطابقت رکھتا ہے اس سے نہ وابستہ ہے نہ اس کا توثیق یافتہ۔ اوقات سے متعلق ہر بات میں ادارے کی اپنی اشاعت ہی مرجع ہے۔

کیا محراب میرے مقام کا سراغ رکھتا ہے؟

نہیں۔ نہ کوئی اینالیٹکس ہے، نہ اشتہاری نیٹ ورک، نہ اکاؤنٹ۔ کوآرڈینیٹس فون سے صرف دو درخواستوں میں باہر جاتے ہیں، اور کوئی بھی ڈویلپر کو نہیں جاتی: AlAdhan کو، جب وہ آپ کے اوقات کا ماخذ ہو، اور خودکار مقام کی صورت میں OpenStreetMap کو، آپ کے قصبے کا نام جاننے کے لیے، تقریباً ایک کلومیٹر تک گول کر کے۔ آلے پر حساب منتخب کریں اور اپنا قصبہ خود درج کریں تو دونوں میں سے کوئی درخواست نہیں بھیجی جاتی۔

کیا میں کچھ انسٹال کیے بغیر محراب استعمال کر سکتا ہوں؟

جی ہاں۔ محراب ویب آپ کے براؤزر میں mihrabhq.github.io پر چلتا ہے: آپ کے مقام کے اوقاتِ نماز، ٹیب کھلا رہنے تک ہر نماز پر اطلاع، اور پورا قرآن۔ ترتیبات اور بک مارکس اسی براؤزر میں رہتے ہیں۔
